<script>
var PIN_KEY = "mtp-control-pin";
var pin = sessionStorage.getItem(PIN_KEY) || "";
var rooms = [], els = {}, cfg = {};

function headers() {
  var h = { "Content-Type": "application/json" };
  if (pin) h["x-control-pin"] = pin;
  return h;
}
function fmt(ms) {
  var s = Math.floor((ms || 0) / 1000);
  return Math.floor(s / 60) + ":" + String(s % 60).padStart(2, "0");
}
function title(id) { return id ? id.charAt(0).toUpperCase() + id.slice(1) : "—"; }

function send(r, command, persona) {
  var e = els[r.base];
  // This PC's slot is /api/room/command; the peer relay is /api/peer/command.
  fetch(r.cmd, { method: "POST", headers: headers(),
      body: JSON.stringify({ command: command, persona: persona || null }) })
    .then(function (x) {
      if (x.status === 401) return lock();
      // A command that does not land has to say so. A silent 404 once made every
      // tap on this page do nothing while the status kept updating happily.
      if (!e) return;
      if (x.ok) { e.cmdErr = ""; return; }
      // Prefer the server's own words - it knows things like a PIN mismatch between rooms.
      return x.json().then(function (b) {
        e.cmdErr = (b && b.error) || "Command not delivered (" + x.status + ").";
      }).catch(function () { e.cmdErr = "Command not delivered (" + x.status + ")."; });
    })
    .then(poll)
    .catch(function () { if (e) e.cmdErr = "Command not delivered - no reply."; poll(); });
}

function card(r) {
  var d = document.createElement("div");
  d.className = "room";
  d.innerHTML =
    '<div class="head"><span class="id"></span><span class="pill"></span><span class="clock"></span></div>' +
    '<div class="who">Next patient</div><div class="picks"></div>' +
    '<div class="acts"><button class="go start">Start</button><button class="stop end">End</button>' +
    '<div class="minor"><button class="skip">Skip</button><button class="shut"></button></div></div>' +
    '<div class="err"></div>';
  document.getElementById("rooms").appendChild(d);
  var e = {
    id: d.querySelector(".id"), pill: d.querySelector(".pill"), clock: d.querySelector(".clock"),
    picks: d.querySelector(".picks"), start: d.querySelector(".start"), end: d.querySelector(".end"),
    skip: d.querySelector(".skip"), shut: d.querySelector(".shut"), err: d.querySelector(".err"),
    buttons: {}
  };
  e.id.textContent = r.label;
  (cfg.rotation || []).forEach(function (id) {
    var b = document.createElement("button");
    b.textContent = title(id);
    b.onclick = function () { send(r, "pick", id); };
    e.picks.appendChild(b);
    e.buttons[id] = b;
  });
  e.start.onclick = function () { send(r, "start"); };
  e.end.onclick = function () { send(r, "end"); };
  e.skip.onclick = function () { send(r, "skip"); };
  e.shut.onclick = function () { send(r, e.shutState === "closed" ? "open" : "close"); };
  return e;
}

function render(r, data) {
  var e = els[r.base];
  var off = !data || data.error === undefined && data.state === undefined;
  if (off) {
    e.pill.className = "pill offline"; e.pill.textContent = "offline";
    e.clock.textContent = ""; e.clock.className = "clock";
    [e.start, e.end, e.skip, e.shut].forEach(function (b) { b.disabled = true; });
    Object.keys(e.buttons).forEach(function (k) { e.buttons[k].disabled = true; });
    e.err.className = "err on"; e.err.textContent = "No reply from this room.";
    return;
  }
  var live = data.state === "live", closed = data.state === "closed";
  e.pill.className = "pill " + data.state;
  e.pill.textContent = data.state;
  e.clock.textContent = live ? fmt(data.elapsedMs) : "";
  e.clock.className = "clock" + (live ? " on" : "");
  e.start.disabled = live || closed;
  e.end.disabled = !live;
  e.skip.disabled = live;
  e.shut.disabled = false;
  e.shutState = data.state;
  e.shut.textContent = closed ? "Reopen stand" : "Close stand";
  Object.keys(e.buttons).forEach(function (k) {
    e.buttons[k].disabled = live;
    e.buttons[k].className = k === data.persona ? "sel" : "";
  });
  var msg = data.error || e.cmdErr || "";
  e.err.className = "err" + (msg ? " on" : "");
  e.err.textContent = msg;
}

function poll() {
  rooms.forEach(function (r) {
    fetch(r.base + "/room", { cache: "no-store", headers: headers() })
      .then(function (x) { return x.ok ? x.json() : null; })
      .then(function (d) { render(r, d); })
      .catch(function () { render(r, null); });
  });
}

function lock() {
  sessionStorage.removeItem(PIN_KEY); pin = "";
  document.getElementById("gate").className = "on";
  document.getElementById("rooms").innerHTML = "";
  document.getElementById("foot").textContent = "";
}

function boot() {
  // Rooms are numbered, not lettered, and Room 1 is always the top card whichever PC is
  // serving this page - so the host's phone reads the same in both rooms.
  var mine = (cfg.roomId || "A").toUpperCase() === "B" ? 2 : 1;
  var local = { base: "/api", cmd: "/api/room/command", label: "Room " + mine };
  rooms = [local];
  if (cfg.hasPeer) {
    var other = { base: "/api/peer", cmd: "/api/peer/command", label: "Room " + (mine === 1 ? 2 : 1) };
    rooms = mine === 1 ? [local, other] : [other, local];
  }
  document.getElementById("rooms").innerHTML = "";
  els = {};
  rooms.forEach(function (r) { els[r.base] = card(r); });
  document.getElementById("foot").textContent = cfg.hasPeer
    ? "Both rooms · relayed through this PC"
    : "This room only — set PEER_URL in .env";
  poll();
  setInterval(poll, 1000);
}

document.getElementById("pinGo").onclick = function () {
  pin = document.getElementById("pin").value.trim();
  sessionStorage.setItem(PIN_KEY, pin);
  document.getElementById("gate").className = "";
  boot();
};
document.getElementById("pin").addEventListener("keydown", function (e) {
  if (e.key === "Enter") document.getElementById("pinGo").click();
});

fetch("/api/config").then(function (r) { return r.json(); }).then(function (c) {
  cfg = c;
  if (c.pinRequired && !pin) { document.getElementById("gate").className = "on"; return; }
  boot();
});
</script>
